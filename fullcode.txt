// FILE: main.cpp

#include <QGuiApplication>
#include <QQmlApplicationEngine>

int main(int argc, char *argv[])
{
    QGuiApplication app(argc, argv);

    QCoreApplication::setOrganizationName("faraz");
    QCoreApplication::setApplicationName("fc");

    QQmlApplicationEngine engine;
    QObject::connect(
        &engine,
        &QQmlApplicationEngine::objectCreationFailed,
        &app,
        []() { QCoreApplication::exit(-1); },
        Qt::QueuedConnection);
    engine.loadFromModule("faraz", "Main");

    return QGuiApplication::exec();
}

Item {
    id: mapContainer

    Layout.preferredHeight: 400

        anchors.fill: parent
        plugin: mapPlugin
        activeMapType: supportedMapTypes.length > 0
                       ? supportedMapTypes[supportedMapTypes.length - 1]
                       : undefined
        center: (df && df.station && df.station.isValid)
                ? df.station
                : QtPositioning.coordinate(0, 0)
        zoomLevel: 16

        MapCircle {
            visible: !!(df && df.station && df.station.isValid)
            center: (df && df.station) ? df.station : QtPositioning.coordinate(0, 0)
            radius: 45
            color: "transparent"
            border.color: Theme.accentColor
            border.width: 2
        }

        MapPolyline {
            visible: !!(df && df.active && df.hasBearing
                        && df.station && df.station.isValid
                        && df.lineEnd && df.lineEnd.isValid)
            line.width: 2
            line.color: Theme.accentColor
            path: (df && df.station && df.lineEnd) ? [df.station, df.lineEnd] : []
        }

        MapQuickItem {
            visible: !!(df && df.station && df.station.isValid)
            coordinate: (df && df.station) ? df.station : QtPositioning.coordinate(0, 0)
            anchorPoint.x: 7
            anchorPoint.y: 7
            sourceItem: Item {
                width: 14
                height: 14
                Rectangle {
                    anchors.fill: parent
                    rotation: 45
                    color: Theme.accentColor
                    border.color: "white"
                    border.width: 2
                }
                Text {
                    x: 18
                    y: -2
                    text: "DF 1"
                    color: "white"
                    font.pixelSize: 11
                    style: Text.Outline
                    styleColor: "black"
                }
            }
        }

        DragHandler {
            id: panHandler
            target: null
            acceptedButtons: Qt.LeftButton
            grabPermissions: PointerHandler.CanTakeOverFromItems | PointerHandler.ApprovesTakeOverByItems

            property real lastX: 0
            property real lastY: 0

            onActiveChanged: {
                if (active) {
                    lastX = translation.x
                    lastY = translation.y
                }
            }

            onTranslationChanged: {
                if (!active)
                    return
                var dx = translation.x - lastX
                var dy = translation.y - lastY
                lastX = translation.x
                lastY = translation.y
                // Dragging right/down moves the viewport left/up (-dx, -dy)
                gisMap.pan(-dx, -dy)
            }
        }
    }

    MapHeatmap {
        anchors.fill: parent
        map: gisMap
        locator: df
        enabled: false
    }

    Canvas {
        id: cornerRadiusOverlay
        anchors.fill: parent
        enabled: false
        onWidthChanged: requestPaint()
        onHeightChanged: requestPaint()

        Connections {
            target: Theme
            function onBackgroundColorChanged() { cornerRadiusOverlay.requestPaint() }
        }

        onPaint: {
            var ctx = getContext("2d")
            ctx.clearRect(0, 0, width, height)

            var r = Math.min(mapContainer.cornerRadius, width / 2, height / 2)
            if (r <= 0)
                return

            ctx.fillStyle = Theme.backgroundColor.toString()

            // Top-left corner
            ctx.beginPath()
            ctx.moveTo(0, 0)
            ctx.lineTo(r, 0)
            ctx.arc(r, r, r, -Math.PI / 2, Math.PI, true)
            ctx.closePath()
            ctx.fill()

            // Top-right corner
            ctx.beginPath()
            ctx.moveTo(width, 0)
            ctx.lineTo(width, r)
            ctx.arc(width - r, r, r, 0, -Math.PI / 2, true)
            ctx.closePath()
            ctx.fill()

            // Bottom-right corner
            ctx.beginPath()
            ctx.moveTo(width, height)
            ctx.lineTo(width - r, height)
            ctx.arc(width - r, height - r, r, Math.PI / 2, 0, true)
            ctx.closePath()
            ctx.fill()

            // Bottom-left corner
            ctx.beginPath()
            ctx.moveTo(0, height)
            ctx.lineTo(0, height - r)
            ctx.arc(r, height - r, r, Math.PI, Math.PI / 2, true)
            ctx.closePath()
            ctx.fill()
        }
    }
}

// FILE: qml/panels/MapHeatmap.qml

import QtQuick
import QtLocation
import faraz

HeatmapItem {
    id: root
    required property Map map

    function updateRect() {
        if (!map) return
        const tl = map.fromCoordinate(root.gridTopLeft, false)
        const br = map.fromCoordinate(root.gridBottomRight, false)
        if (isNaN(tl.x) || isNaN(br.x)) return
        viewRect = Qt.rect(tl.x, tl.y, br.x - tl.x, br.y - tl.y)
    }

    Connections {
        target: root.map
        function onCenterChanged()    { root.updateRect() }
        function onZoomLevelChanged() { root.updateRect() }
        function onWidthChanged()     { root.updateRect() }
        function onHeightChanged()    { root.updateRect() }
        function onMapReadyChanged()  { root.updateRect() }
    }

    onGridChanged: updateRect()
    Component.onCompleted: updateRect()
}

import QtQuick
import QtQuick.Layouts

Rectangle {
    id: root

    readonly property color accent: Theme.accentColor
    readonly property color surface: Theme.inputColor
    readonly property color surfaceHover: Theme.hoverColor
    readonly property color outline: Theme.outlineColor
    readonly property color textIdle: Theme.textMuted

    readonly property var unitFactors: [1, 1e3, 1e6, 1e9]
    readonly property var spans: [50e3, 100e3, 200e3, 500e3, 1e6, 5e6, 10e6]
    readonly property var spanLabels: ["50 kHz", "100 kHz", "200 kHz", "500 kHz", "1 MHz", "5 MHz", "10 MHz"]

    property var stationState: null
    signal tuningRequested(real centerFrequencyHz, real spanHz, bool dfMode)

    color: Theme.panelColor
    radius: 8
    border.width: 1
    border.color: Theme.borderColor

    function currentCenterHz() {
        const value = parseFloat(freqField.text)
        if (isNaN(value))
            return root.stationState ? root.stationState.centerFrequency : 100e6
        return Math.max(0, value * unitFactors[unitBox.currentIndex])
    }

    function currentSpanHz() {
        const idx = Math.round(freqSlider.value)
        return (idx >= 0 && idx < spans.length)
               ? spans[idx]
               : (root.stationState ? root.stationState.span : 1e6)
    }

    function sendCurrentTuning() {
        root.tuningRequested(currentCenterHz(), currentSpanHz(), dfSwitch.checked)
    }

    function syncFromState() {
        if (!stationState)
            return

        const freq = stationState.centerFrequency
        let idx = 2
        for (let i = unitFactors.length - 1; i >= 0; --i) {
            if (freq >= unitFactors[i]) {
                idx = i
                break
            }
        }
        unitBox.currentIndex = idx
        freqField.text = String(freq / unitFactors[idx])

        const s = spans.indexOf(stationState.span)
        freqSlider.value = s >= 0 ? s : 4
        dfSwitch.checked = stationState.dfMode
    }

    Component.onCompleted: syncFromState()
    onStationStateChanged: syncFromState()

    Connections {
        target: root.stationState
        ignoreUnknownSignals: true
        function onCenterFrequencyChanged() { root.syncFromState() }
        function onSpanChanged() { root.syncFromState() }
        function onDfModeChanged() { root.syncFromState() }
    }

import QtQuick
import faraz

Rectangle {
    id: root
    color: Theme.panelColor
    radius: 10
    border.width: 1
    border.color: Theme.borderColor

    property alias minDb: wave.minDb
    property alias maxDb: wave.maxDb
    property alias source: wave.source
    property alias wave: wave

    function formatHz(hz) {
        const a = Math.abs(hz)
        if (a >= 1e9) return (hz / 1e9).toFixed(4) + " GHz"
        if (a >= 1e6) return (hz / 1e6).toFixed(4) + " MHz"
        if (a >= 1e3) return (hz / 1e3).toFixed(2) + " kHz"
        return hz.toFixed(0) + " Hz"
    }

    Item {
        id: plot
        anchors.fill: parent
        anchors.margins: 12
        anchors.leftMargin: 44
        anchors.bottomMargin: 28

        // horizontal grid + dB labels (every 10 dB)
        Repeater {
            model: Math.round((root.maxDb - root.minDb) / 10) + 1
            Item {
                required property int index
                width: plot.width
                y: plot.height * index / (model_count - 1)
                readonly property int model_count: Math.round((root.maxDb - root.minDb) / 10) + 1

                Rectangle { width: parent.width; height: 1; color: Theme.gridColor }
                Text {
                    x: -40; y: -height / 2

QVariant StationModel::data(const QModelIndex &index, int role) const

    if (!index.isValid() || index.row() < 0 || index.row() >= m_entries.size())
        return {};

    const Entry &entry = m_entries.at(index.row());
    switch (role) {
    case NameRole:
        return entry.state ? entry.state->name() : QString();
    case TcpPortRole:
        return entry.state ? entry.state->tcpPort() : 0;
    case StateRole:
        return entry.state ? QVariant::fromValue(entry.state) : QVariant();
    default:
        return {};
    }
}

    emit currentStationChanged();

    return stationAt(m_currentIndex);

        return nullptr;
    return m_entries.at(index).state;

void StationModel::setBasePort(int port)
{
    if (port < 1 || port > 65535 || m_basePort == port)
        return;

    m_basePort = port;
    for (int i = 0; i < m_entries.size(); ++i) {
        const quint16 p = static_cast<quint16>(m_basePort + i);
        m_entries[i].state->setTcpPort(p);
        m_entries[i].stream->listen(p);
    }
    emit basePortChanged();
}

void StationModel::addStation(const QString &name)
{
    const int row = static_cast<int>(m_entries.size());
    const quint16 port = static_cast<quint16>(m_basePort + row);
    const QString resolvedName = name.isEmpty()
                                     ? QStringLiteral("Station %1").arg(row + 1)
                                     : name;

    beginInsertRows(QModelIndex(), row, row);

    auto *state = new StationState(this);
    state->setName(resolvedName);
    state->setTcpPort(port);

    auto *stream = createLink();
    connect(stream, &ILink::packetReceived,
            state, &StationState::applyPacket);
    stream->listen(port);

    m_entries.push_back(Entry{ state, stream });

    endInsertRows();

    setCurrentIndex(row);
}

bool StationModel::tuneCurrentStation(double centerFrequencyHz, double spanHz, bool dfMode)
{
    return tuneStation(currentStation(), centerFrequencyHz, spanHz, dfMode);
}

bool StationModel::tuneStation(StationState *station, double centerFrequencyHz, double spanHz, bool dfMode)
{
    if (!station)
        return false;

    for (const Entry &entry : m_entries) {
        if (entry.state == station && entry.stream) {
            return entry.stream->sendTuning(centerFrequencyHz, spanHz, dfMode);
        }
    }
    return false;
}

    if (port < 0 || port > 65535 || m_tcpPort == port)
        return;
    m_tcpPort = port;
    emit tcpPortChanged();

void StationState::applyPacket(const faraz::Packet &pkt)
{
    if ((pkt.flags & faraz::PacketFlagHasCoord) != 0
        && std::isfinite(pkt.latitude) && std::isfinite(pkt.longitude)) {
        const QGeoCoordinate coord(pkt.latitude, pkt.longitude);
        if (coord.isValid() && m_station != coord) {
            m_station = coord;
            emit stationChanged();
        }
    }

    bool axisUpdated = false;
    if (pkt.centerFrequencyHz > 0.0 && std::isfinite(pkt.centerFrequencyHz)
        && !qFuzzyCompare(m_center + 1.0, pkt.centerFrequencyHz + 1.0)) {
        m_center = pkt.centerFrequencyHz;
        emit centerFrequencyChanged();
        axisUpdated = true;
    }

    if (pkt.spanHz > 0.0 && std::isfinite(pkt.spanHz)
        && !qFuzzyCompare(m_span, pkt.spanHz)) {
        m_span = pkt.spanHz;
        emit spanChanged();
        axisUpdated = true;
    }

    const bool pktDfMode = (pkt.flags & faraz::PacketFlagDfMode) != 0;
    if (m_dfMode != pktDfMode) {
        m_dfMode = pktDfMode;
        if (!m_dfMode) {
            resetDf();
        }
        emit dfModeChanged();
        emit activeChanged();
    }

    if ((pkt.flags & faraz::PacketFlagResetAxis) != 0 || axisUpdated) {
        emit axisChanged();
    }

    if ((pkt.flags & faraz::PacketFlagResetDf) != 0) {
        resetDf();
    }

    if ((pkt.flags & faraz::PacketFlagHasSpec) != 0 && pkt.binCount > 0) {
        const int count = static_cast<int>(
            std::min<quint32>(pkt.binCount, static_cast<quint32>(faraz::kMaxPacketBins)));
        QVector<float> dbm(count);
        for (int i = 0; i < count; ++i) {
            const float v = pkt.bins[i];
            dbm[i] = std::isfinite(v) ? v : -120.f;
        }
        emit frameReady(dbm);
    }

    if ((pkt.flags & faraz::PacketFlagHasDf) != 0
        && std::isfinite(pkt.bearingDeg) && std::isfinite(pkt.levelDbm)) {
        if (m_dfMode) {
            m_bearing = pkt.bearingDeg;
            m_hasBearing = true;
            emit bearingChanged();
        }
        emit bearingMeasured(pkt.bearingDeg, pkt.levelDbm);
    }
}

QGeoCoordinate StationState::lineEnd() const
{
    if (!m_station.isValid())
        return QGeoCoordinate();
    return m_station.atDistanceAndAzimuth(kLineLength, m_bearing);
}

void StationState::setCenterFrequency(double hz)
{
    hz = std::max(0.0, hz);
    if (qFuzzyCompare(m_center + 1.0, hz + 1.0))
        return;
    m_center = hz;
    emit centerFrequencyChanged();
    emit axisChanged();
    emit tuningChanged();
}

void StationState::setSpan(double hz)
{
    hz = std::max(1.0, hz);
    if (qFuzzyCompare(m_span, hz))
        return;
    m_span = hz;
    emit spanChanged();
    emit axisChanged();
    emit tuningChanged();
}

void StationState::setDfMode(bool on)
{
    if (m_dfMode == on)
        return;
    m_dfMode = on;
    if (!m_dfMode)
        resetDf();
    emit dfModeChanged();
    emit activeChanged();
    emit tuningChanged();
}

void StationState::resetDf()
{
    if (!m_hasBearing && qFuzzyIsNull(m_bearing))
        return;
    m_hasBearing = false;
    m_bearing = 0.0;
    emit bearingChanged();
}

TcpLink::TcpLink(QObject *parent)
    : ILink(parent)

    connect(&m_server, &QTcpServer::newConnection,
            this, &TcpLink::onNewConnection);

TcpLink::~TcpLink()

    close();
}

bool TcpLink::listen(quint16 port)
{
    close();
    m_port = port;

    if (m_server.listen(QHostAddress::AnyIPv4, m_port)) {
        qInfo() << "TcpLink listening on port" << m_port;
        return true;
    }

    qWarning() << "TcpLink failed to listen on port" << m_port << m_server.errorString();
    return false;
}

void TcpLink::close()
{
    if (m_client) {
        m_client->disconnect(this);
        m_client->disconnectFromHost();
        m_client->abort();
        m_client->deleteLater();
        m_client = nullptr;
    }
    m_buffer.clear();
    m_server.close();
}

bool TcpLink::hasClient() const
{
    return m_client && m_client->state() == QAbstractSocket::ConnectedState;
}

void TcpLink::onNewConnection()
{
    QTcpSocket *next = m_server.nextPendingConnection();
    if (!next)
        return;

    const bool hadClient = (m_client != nullptr);
    if (m_client) {
        m_client->disconnect(this);
        m_client->abort();
        m_client->deleteLater();
        m_client = nullptr;
        m_buffer.clear();
    }

    m_client = next;
    m_client->setSocketOption(QAbstractSocket::LowDelayOption, 1);

    connect(m_client, &QTcpSocket::readyRead,
            this, &TcpLink::onReadyRead);
    connect(m_client, &QTcpSocket::disconnected,
            this, &TcpLink::onDisconnected);

    qInfo() << "TcpLink accepted connection from"
            << m_client->peerAddress().toString() << m_client->peerPort();

    if (!hadClient) {
        emit clientConnected();
    }
}

void TcpLink::onDisconnected()
{
    if (!m_client)
        return;

    m_client->deleteLater();
    m_client = nullptr;
    m_buffer.clear();
    emit clientDisconnected();
}

void TcpLink::onReadyRead()
{
    if (!m_client)
        return;

    m_buffer.append(m_client->readAll());
    processBuffer();
}

void TcpLink::processBuffer()
{
    constexpr int headerBytes = static_cast<int>(sizeof(quint32));
    constexpr quint32 headerOnlyBytes = static_cast<quint32>(offsetof(faraz::Packet, bins));

    while (m_buffer.size() >= headerBytes) {
        const auto *raw = reinterpret_cast<const uchar *>(m_buffer.constData());
        const quint32 frameSize = (static_cast<quint32>(raw[0]) << 24)
                                  | (static_cast<quint32>(raw[1]) << 16)
                                  | (static_cast<quint32>(raw[2]) << 8)
                                  | (static_cast<quint32>(raw[3]));

        if (frameSize == 0 || frameSize > kMaxTcpFrameBytes) {
            qWarning() << "Invalid TCP frame size:" << frameSize << "- clearing buffer";
            m_buffer.clear();
            return;
        }

        const int totalNeeded = headerBytes + static_cast<int>(frameSize);
        if (m_buffer.size() < totalNeeded) {
            return;
        }

        if (frameSize >= headerOnlyBytes && frameSize <= sizeof(faraz::Packet)) {
            faraz::Packet pkt{};
            std::memcpy(&pkt, raw + headerBytes, static_cast<size_t>(frameSize));

            const quint32 maxPossibleBins =
                (frameSize - headerOnlyBytes) / static_cast<quint32>(sizeof(float));
            if (pkt.binCount > maxPossibleBins) {
                pkt.binCount = maxPossibleBins;
            }

            if (pkt.magic == faraz::kPacketMagic && pkt.version == faraz::kPacketVersion) {
                emit packetReceived(pkt);
            } else {
                qWarning() << "Invalid Packet magic/version";
            }
        } else {
            qWarning() << "Unexpected Packet payload size:" << frameSize;
        }

        m_buffer.remove(0, totalNeeded);
    }
}

bool TcpLink::sendTuning(double centerFrequencyHz, double spanHz, bool dfMode)
{
    if (!hasClient()) {
        qWarning() << "TcpLink::sendTuning: no connected client on port" << m_port;
        return false;
    }

    QJsonObject obj;
    obj.insert(QStringLiteral("type"), QStringLiteral("tune"));
    obj.insert(QStringLiteral("centerFrequencyHz"), centerFrequencyHz);
    obj.insert(QStringLiteral("spanHz"), spanHz);
    obj.insert(QStringLiteral("dfMode"), dfMode);

    QByteArray line = QJsonDocument(obj).toJson(QJsonDocument::Compact);
    line.append('\n');

    const qint64 written = m_client->write(line);
    return written == line.size();
}

inline double wrap360(double deg)
{
    double r = std::fmod(deg, 360.0);
    return r < 0.0 ? r + 360.0 : r;
}

inline double angDiffDeg(double a, double b)
{
    double d = std::fmod(a - b + 540.0, 360.0) - 180.0;
    return d < -180.0 ? d + 360.0 : d;
}

// Jet-style ramp: blue -> cyan -> green -> yellow -> red. Returns PREMULTIPLIED ARGB.
inline QRgb heatColor(float t, float intensity)
{
    const float v = std::clamp(t * intensity, 0.f, 1.f);
    if (v <= 0.02f)
        return 0;   // fully transparent

    float r, g, b;
    if (v < 0.25f)      { r = 0.f;                  g = 4.f * v;                 b = 1.f; }
    else if (v < 0.50f) { r = 0.f;                  g = 1.f;                     b = 1.f - 4.f * (v - 0.25f); }
    else if (v < 0.75f) { r = 4.f * (v - 0.50f);    g = 1.f;                     b = 0.f; }
    else                { r = 1.f;                  g = 1.f - 4.f * (v - 0.75f); b = 0.f; }

    const int a = static_cast<int>(70.f + 165.f * v);   // stays visible at the low end
    return qPremultiply(qRgba(static_cast<int>(r * 255.f),
                              static_cast<int>(g * 255.f),
                              static_cast<int>(b * 255.f),
                              a));
}
} // namespace

    const QRectF r = m_viewRect;
    if (!std::isfinite(r.x()) || !std::isfinite(r.y())
        || !std::isfinite(r.width()) || !std::isfinite(r.height())
        || r.width() <= 0.0 || r.height() <= 0.0) {
        return;
    }

    painter->save();
    painter->setRenderHint(QPainter::SmoothPixmapTransform, true);
    painter->drawImage(r, m_image);
    painter->restore();
}

double HeatmapItem::metersPerDegLon() const
{
    const double lat = m_station.isValid() ? m_station.latitude() : 0.0;
    const double latRad = lat * kDeg2Rad;
    return std::max(1.0, kMetersPerDegLat * std::cos(latRad));
}

QGeoCoordinate HeatmapItem::gridTopLeft() const
{
    const double dLat = m_extent / kMetersPerDegLat;
    const double dLon = m_extent / metersPerDegLon();
    return QGeoCoordinate(m_station.latitude() + dLat, m_station.longitude() - dLon);
}

QGeoCoordinate HeatmapItem::gridBottomRight() const
{
    const double dLat = m_extent / kMetersPerDegLat;
    const double dLon = m_extent / metersPerDegLon();
    return QGeoCoordinate(m_station.latitude() - dLat, m_station.longitude() + dLon);
}

void HeatmapItem::setGridExtent(double meters)
{
    meters = std::max(50.0, meters);
    if (qFuzzyCompare(m_extent, meters)) return;
    m_extent = meters;
    emit gridChanged();
}

void HeatmapItem::reset()
{
    std::fill(m_hist.begin(), m_hist.end(), 0.f);
    m_hasBearing = false;
    m_bearing = 0.0;
    m_outliers = 0;
    publishGrid();
}

void HeatmapItem::rebuildCellMap()
{
    const double half = 0.5 * kGrid;
    for (int r = 0; r < kGrid; ++r) {
        const double dy = (half - (r + 0.5)) / half;
        for (int c = 0; c < kGrid; ++c) {
            const double dx = ((c + 0.5) - half) / half;
            const double dist2 = dx * dx + dy * dy;

            m_cellDist[r * kGrid + c] = static_cast<float>(std::sqrt(std::min(dist2, 1.0)));

            if (dist2 < 1e-6 || dist2 > 1.0) {
                m_cellPos[r * kGrid + c] = -1.f;
                continue;
            }
            const double azDeg = wrap360(std::atan2(dx, dy) * kRad2Deg);
            m_cellPos[r * kGrid + c] = static_cast<float>(azDeg * kBins / 360.0);   // fractional
        }
    }
}

void HeatmapItem::pushBearing(double rawDeg)
{
    const double b = wrap360(rawDeg);

    // Smoothed bearing (used for outlier gating / map line) - unchanged logic
    if (!m_hasBearing) {
        m_bearing = b;
        m_hasBearing = true;
        m_outliers = 0;
    } else if (std::abs(angDiffDeg(b, m_bearing)) > 35.0 && ++m_outliers < 4) {
        return;     // reject isolated outliers
    } else {
        const double alpha = (m_outliers >= 4) ? 0.6 : 0.25;
        const double delta = angDiffDeg(b, m_bearing);
        m_bearing = wrap360(m_bearing + alpha * delta);
        m_outliers = 0;
    }

    // Histogram of RAW bearings: the spread of the measurements becomes the cone width.
    constexpr float kDecay = 0.985f;        // was 0.94: remember more samples
    for (float &v : m_hist)
        v *= kDecay;

    const double centerBin = (b / 360.0) * kBins;   // raw, not the smoothed m_bearing
    constexpr double kSigmaBins = 5.0;              // 2.5 deg at 0.5 deg/bin
    constexpr int kHalf = 18;
    for (int k = -kHalf; k <= kHalf; ++k) {
        int idx = (static_cast<int>(std::llround(centerBin)) + k) % kBins;
        if (idx < 0) idx += kBins;
        const double d = (idx - centerBin);
        const double wrappedD = d - kBins * std::llround(d / kBins);
        m_hist[idx] += static_cast<float>(std::exp(-0.5 * (wrappedD * wrappedD) / (kSigmaBins * kSigmaBins)));
    }
}

void HeatmapItem::publishGrid()
{
    float peak = 0.f;
    for (float v : m_hist)
        peak = std::max(peak, v);

    std::fill(m_values.begin(), m_values.end(), quint8(0));
    if (peak > 1e-4f) {
        const float invPeak = 255.f / peak;
        const int total = kGrid * kGrid;
        for (int i = 0; i < total; ++i) {
            const float pos = m_cellPos[i];
            if (pos < 0.f) continue;

            const int i0 = static_cast<int>(pos);          // pos >= 0, so truncation == floor
            const float f = pos - static_cast<float>(i0);
            const int b0 = i0 % kBins;
            const int b1 = (i0 + 1) % kBins;
            const float h = m_hist[b0] + f * (m_hist[b1] - m_hist[b0]);   // interpolate between bins

            const float fall = 1.0f - 0.55f * m_cellDist[i];              // fades along the ray
            const float val = h * invPeak * fall;
            m_values[i] = static_cast<quint8>(std::clamp(static_cast<int>(val + 0.5f), 0, 255));
        }
    }

    rebuildImage();
    update();
}

void HeatmapItem::rebuildLut()
{
    m_lut.resize(256);
    for (int v = 0; v < 256; ++v)
        m_lut[v] = heatColor(static_cast<float>(v) / 255.f, m_intensity);
}

void HeatmapItem::rebuildImage()
{
    const QRgb *lut = m_lut.constData();
    for (int y = 0; y < kGrid; ++y) {
        auto *scan = reinterpret_cast<QRgb *>(m_image.scanLine(y));
        const quint8 *src = m_values.constData() + y * kGrid;
        for (int x = 0; x < kGrid; ++x)
            scan[x] = lut[src[x]];
    }
}

void HeatmapItem::onBearingMeasured(double bearingDeg, double /*levelDb*/)
{
    if (m_locator && m_locator->station().isValid() && m_station != m_locator->station()) {
        m_station = m_locator->station();
        emit gridChanged();
    }
    if (!m_locator || !m_locator->dfMode())
        return;

    pushBearing(bearingDeg);
    publishGrid();
}

void HeatmapItem::setLocator(StationState *l)
{
    if (m_locator == l) return;

    for (const auto &conn : std::as_const(m_conns))
        QObject::disconnect(conn);
    m_conns.clear();

    m_locator = l;
    if (m_locator) {
        if (m_locator->station().isValid())
            m_station = m_locator->station();

        m_conns << connect(m_locator, &StationState::bearingMeasured,
                           this, &HeatmapItem::onBearingMeasured);
        m_conns << connect(m_locator, &StationState::stationChanged, this, [this]() {
            if (m_locator && m_locator->station().isValid() && m_station != m_locator->station()) {
                m_station = m_locator->station();
                reset();
                emit gridChanged();
            }
        });
        m_conns << connect(m_locator, &StationState::axisChanged,
                           this, &HeatmapItem::reset);
        m_conns << connect(m_locator, &StationState::dfModeChanged, this, [this]() {
            if (m_locator && !m_locator->dfMode())
                reset();
        });

        reset();
    } else {
        reset();
    }
    emit locatorChanged();
}

void HeatmapItem::setViewRect(const QRectF &r)
{
    if (m_viewRect == r) return;
    m_viewRect = r;
    emit viewRectChanged();
    update();
}

void HeatmapItem::setIntensity(float v)
{
    if (qFuzzyCompare(m_intensity, v)) return;
    m_intensity = v;
    rebuildLut();
    rebuildImage();
    emit intensityChanged();
    update();
}

void SpectrumWaveItem::paint(QPainter *painter)
{
    const int count = static_cast<int>(m_data.size());
    if (count < 2)
        return;

    const QRectF rect = boundingRect();
    if (rect.width() <= 0.0 || rect.height() <= 0.0)
        return;

    const float spanDb = m_maxDb - m_minDb;
    if (qFuzzyIsNull(spanDb))
        return;

    painter->save();
    painter->setRenderHint(QPainter::Antialiasing, true);

    QPen pen(m_color, 1.0);
    pen.setCosmetic(true);
    pen.setJoinStyle(Qt::RoundJoin);
    pen.setCapStyle(Qt::RoundCap);
    painter->setPen(pen);

    QPolygonF points;
    points.reserve(count);

    const qreal x0 = rect.left();
    const qreal yBottom = rect.bottom();
    const qreal width = rect.width();
    const qreal height = rect.height();
    const qreal invSpan = 1.0 / static_cast<qreal>(spanDb);
    const qreal stepX = width / static_cast<qreal>(count - 1);

    for (int i = 0; i < count; ++i) {
        const qreal x = x0 + static_cast<qreal>(i) * stepX;
        const qreal normY = std::clamp((static_cast<qreal>(m_data[i]) - static_cast<qreal>(m_minDb)) * invSpan,
                                       0.0, 1.0);
        const qreal y = yBottom - normY * height;
        points.append(QPointF(x, y));
    }

    painter->drawPolyline(points);
    painter->restore();
}

// waterfallitem.cpp
#include "waterfallitem.h"
#include <QPainter>
#include <QtMath>
#include <algorithm>
#include <cmath>

namespace {
// Smooth 5-stop thermal/turbo-like palette (black -> blue -> cyan -> yellow -> white)
QRgb samplePalette(float t)
{
    t = std::clamp(t, 0.f, 1.f);

    struct Stop {
        float pos;
        int r, g, b;
    };
    static constexpr Stop stops[] = {
        { 0.00f,   0,   0,  10 },
        { 0.25f,  15,  40, 150 },
        { 0.55f,   0, 200, 210 },
        { 0.82f, 255, 225,   0 },
        { 1.00f, 255, 255, 255 }
    };
    constexpr int numStops = static_cast<int>(sizeof(stops) / sizeof(stops[0]));

    for (int i = 0; i < numStops - 1; ++i) {
        if (t <= stops[i + 1].pos) {
            const float localT = (t - stops[i].pos) / (stops[i + 1].pos - stops[i].pos);
            const int r = static_cast<int>(std::lround(stops[i].r + localT * (stops[i + 1].r - stops[i].r)));
            const int g = static_cast<int>(std::lround(stops[i].g + localT * (stops[i + 1].g - stops[i].g)));
            const int b = static_cast<int>(std::lround(stops[i].b + localT * (stops[i + 1].b - stops[i].b)));
            return qRgb(std::clamp(r, 0, 255),
                        std::clamp(g, 0, 255),
                        std::clamp(b, 0, 255));
        }
    }
    return qRgb(255, 255, 255);
}
} // namespace

WaterfallItem::WaterfallItem(QQuickItem *parent)
    : QQuickPaintedItem(parent)
    , m_history(kTexW, kTexH, QImage::Format_Grayscale8)
    , m_colored(kTexW, kTexH, QImage::Format_RGB32)
{
    m_history.fill(0);
    m_colored.fill(Qt::black);
    m_head = kTexH - 1;
    rebuildColormap();
}

void WaterfallItem::rebuildColormap()
{
    m_colormap.resize(256);

    const float spanDb = m_maxDb - m_minDb;
    if (qFuzzyIsNull(spanDb)) {
        m_colormap.fill(qRgb(0, 0, 0));
        recolorAll();
        return;
    }

    constexpr float storeSpan = kStoreMax - kStoreMin;
    for (int v = 0; v < 256; ++v) {
        // Convert stored 8-bit value back to dBm, then normalize into [m_minDb, m_maxDb]
        const float db = kStoreMin + (static_cast<float>(v) / 255.f) * storeSpan;
        const float norm = (db - m_minDb) / spanDb;
        m_colormap[v] = samplePalette(norm);
    }
    recolorAll();   // palette changed: recolor whole history once (not per frame)
}

void WaterfallItem::recolorRow(int row)
{
    if (m_colormap.size() != 256)
        return;
    const uchar *src = m_history.constScanLine(row);
    QRgb *dst = reinterpret_cast<QRgb *>(m_colored.scanLine(row));
    const QRgb *lut = m_colormap.constData();
    for (int x = 0; x < kTexW; ++x)
        dst[x] = lut[src[x]];
}

void WaterfallItem::recolorAll()
{
    for (int r = 0; r < kTexH; ++r)
        recolorRow(r);
}

void WaterfallItem::clearHistory()
{
    m_history.fill(0);
    m_colored.fill(Qt::black);
    m_head = kTexH - 1;
    m_filledRows = 0;
}

void WaterfallItem::paint(QPainter *painter)
{
    const QRectF targetRect = boundingRect();
    if (targetRect.width() <= 0.0 || targetRect.height() <= 0.0)
        return;

    painter->save();
    painter->fillRect(targetRect, Qt::black);

    if (m_filledRows > 0) {
        const int start = (m_head + 1) % kTexH;
        const int a = std::min(m_filledRows, kTexH - start);   // rows before the wrap
        const int b = m_filledRows - a;                        // rows after the wrap
        const qreal rowHeight = targetRect.height() / static_cast<qreal>(kTexH);

        painter->setRenderHint(QPainter::SmoothPixmapTransform, true);

        painter->drawImage(QRectF(targetRect.left(), targetRect.top(),
                                  targetRect.width(), rowHeight * a),
                           m_colored,
                           QRectF(0, start, kTexW, a));

        if (b > 0) {
            painter->drawImage(QRectF(targetRect.left(), targetRect.top() + rowHeight * a,
                                      targetRect.width(), rowHeight * b),
                               m_colored,
                               QRectF(0, 0, kTexW, b));
        }
    }

    painter->restore();
}

void WaterfallItem::setSource(StationState *s)
{
    if (m_source == s)
        return;

    QObject::disconnect(m_frameConn);
    QObject::disconnect(m_tuneConn);
    m_source = s;

    clearHistory();

    if (m_source) {
        m_frameConn = connect(m_source, &StationState::frameReady,
                              this, &WaterfallItem::onFrame);
        m_tuneConn  = connect(m_source, &StationState::axisChanged,
                             this, &WaterfallItem::onTuningChanged);
    }
    emit sourceChanged();
    update();
}

void WaterfallItem::onTuningChanged()
{
    clearHistory();
    update();
}

void WaterfallItem::onFrame(const QVector<float> &dbm)
{
    const int n = static_cast<int>(dbm.size());
    if (n < 2)
        return;

    uchar *dstRow = m_history.scanLine(m_head);
    constexpr float invRange = 255.f / (kStoreMax - kStoreMin);

    for (int x = 0; x < kTexW; ++x) {
        const int i0 = static_cast<int>(static_cast<qint64>(x) * n / kTexW);
        const int i1 = std::min(n, std::max(i0 + 1, static_cast<int>(static_cast<qint64>(x + 1) * n / kTexW)));
        float m = dbm[i0];
        for (int i = i0 + 1; i < i1; ++i)
            m = std::max(m, dbm[i]);

        const float t = (m - kStoreMin) * invRange;
        dstRow[x] = static_cast<uchar>(std::clamp(t + 0.5f, 0.f, 255.f));
    }

    recolorRow(m_head);                              // colorize ONLY the new row

    m_head = (m_head - 1 + kTexH) % kTexH;           // decrement: newest is always head+1
    m_filledRows = std::min(m_filledRows + 1, kTexH);
    update();
}

void WaterfallItem::setMinDb(float v)
{
    if (qFuzzyCompare(m_minDb, v))
        return;
    m_minDb = v;
    rebuildColormap();
    emit minDbChanged();
    update();
}

void WaterfallItem::setMaxDb(float v)
{
    if (qFuzzyCompare(m_maxDb, v))
        return;
    m_maxDb = v;
    rebuildColormap();
    emit maxDbChanged();
    update();
}
